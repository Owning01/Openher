#### Búsqueda profunda de procesos ocultos
```
# Verificar si hay procesos ocultos via difference entre /proc y ps
for pid in $(ls /proc/ | grep -E '^[0-9]+$'); do
  if ! ps -p $pid >/dev/null 2>&1; then
    echo "Proceso OCULTO: PID $pid"
  fi
done

# Verificar comandos con strings largos en /proc
for pid in $(ls /proc/ | grep -E '^[0-9]+$' | head -50); do
  cmdline=$(cat /proc/$pid/cmdline 2>/dev/null | tr '' ' ')
  [ -n "$cmdline" ] && echo "PID $pid: $cmdline"
done

# Buscar procesos con sockets abiertos
for pid in $(ls /proc/ | grep -E '^[0-9]+$'); do
  if [ -d /proc/$pid/fd ]; then
    for fd in $(ls /proc/$pid/fd/ 2>/dev/null); do
      link=$(readlink /proc/$pid/fd/$fd 2>/dev/null)
      echo "$link" | grep -q "socket:" && echo "PID $pid tiene socket abierto: $link"
    done
  fi
done
```
